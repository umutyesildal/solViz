with t0 as (
    select decoded_instruction:args:side::string as side
    , decoded_instruction:args:weight::int * power(10, -6) as weight
    , signers[0]::string as voter
    , value:pubkey::string AS proposal
    , value:name::string AS name
    , block_timestamp
    from solana.core.fact_decoded_instructions i
    , LATERAL FLATTEN(input => decoded_instruction:accounts)
    where block_timestamp >= '2024-12-04'
        and i.program_id in (
            'GovaE4iu227srtG2s3tZzB4RmWBzw8sTwrCLZz7kN7rY'
        )
        and name = 'proposal'
        and proposal = 'xMLsw7zzBfRXNiQQo42aUohRsibgrmWcPt2mD8HdUUr'
        and side is not null
        and side > 0
        and weight is not null
        and event_type = 'setVote'
        qualify(
            row_number() over (partition by voter order by block_timestamp desc) = 1
        )
)
select proposal
, side
, case 
    when side = '1' then 'Yes, I am comfortable with this proposal.'
    when side = '2' then 'No, I am not comfortable with this proposal.'
  end as options
, round(sum(weight)) as votes
, count(1) as n_voters
, max(block_timestamp)
, min(block_timestamp)
from t0
where weight > 0
group by 1, 2, 3
order by votes desc

WITH price AS (
SELECT
  hour::date AS date1,
  CASE
    WHEN token_address = 'So11111111111111111111111111111111111111112' THEN 'So11111111111111111111111111111111111111111'
    ELSE token_address
  END AS token_address,
  AVG(price) AS price
FROM solana.price.ez_prices_hourly
  -- WHERE token_address = ''
GROUP BY 1,2
),

base AS (
SELECT
  block_timestamp,
  tx_id,
  mint,
  purchaser,
  mint_price * price AS mint_price,
  mint_currency,
  program_id
FROM solana.nft.fact_nft_mints m JOIN price p 
ON m.block_timestamp::date = p.date1 AND m.mint_currency = p.token_address
  WHERE program_id IN (
  'BGUMAp9Gq7iTEuizy4pqaxsTyUCBK68MDfK752saRPUY',
  'CndyV3LdqHUfDLmE5naZjVN8rBZz4tqhdefbAnjHG3JR',
  'metaqbxxUerdq28cj1RbAWkYQm3ybzjb6a8bt518x1s'
)
  AND succeeded = 'TRUE'
-- having mint_price < 1 --excluding liquidity pool nft mint prices
)

SELECT
  DATE_TRUNC(month, block_timestamp) AS timespan,
  COUNT(DISTINCT tx_id) AS mint_transactions,
  SUM(mint_transactions) OVER (ORDER BY timespan) AS cummul_mint_transactions,
  COUNT(DISTINCT mint) AS NFT_mints,
  SUM(NFT_mints) OVER (ORDER BY timespan) AS cummul_NFT_mints,
  COUNT(DISTINCT purchaser) AS minters,
  SUM(minters) OVER (ORDER BY timespan) AS cummul_minters,
  ROUND(SUM(
      CASE
        WHEN mint_price < 1 THEN mint_price
        ELSE 0
      END), 2) AS total_mint_price,
  SUM(total_mint_price) OVER (ORDER BY timespan) AS cummul_mint_price,
  ROUND(AVG(
      CASE
        WHEN mint_price < 1 THEN mint_price
        ELSE 0
      END), 7) AS avg_mint_price
FROM base
GROUP BY 1--, 2 
ORDER BY timespan

WITH vote AS (
  SELECT
    signers[0] AS voter,
    decoded_instruction:accounts[1]:pubkey AS proposal,
    decoded_instruction:args:side AS vote_choice,
    decoded_instruction:args:weight / pow(10, 6) AS vote_weight,
    block_timestamp,
    tx_id
  FROM
    solana.core.ez_events_decoded
  WHERE 1 = 1
    AND block_timestamp::date >= '2024-12-04'
    AND program_id = 'GovaE4iu227srtG2s3tZzB4RmWBzw8sTwrCLZz7kN7rY'
    AND event_type = 'setVote'
    AND proposal = 'xMLsw7zzBfRXNiQQo42aUohRsibgrmWcPt2mD8HdUUr'
    AND succeeded
),
latest_vote AS (
  SELECT
    voter,
    MAX(block_timestamp) AS latest
  FROM
    vote
  GROUP BY
    voter
)
SELECT
  v.voter AS "Wallet Address",
  v.proposal AS "Proposal ID",
  v.block_timestamp AS "Vote Timestamp",
  1 as "Vote Count"
FROM
  vote v
JOIN
  latest_vote lv ON v.voter = lv.voter AND v.block_timestamp = lv.latest
ORDER BY
  v.vote_weight DESC
LIMIT 10000000 -- Large limit to get all remaining rows
OFFSET 100000; -- Skips the first 100,000 rows

with date_filter as (
  select
    dateadd(day, -30, current_date()) as start_date,  -- defaulting to last 30 days
    current_date() as end_date
),

tx_list as(
  select 
    ft.tx_id,
    ft.block_id,
    ft.block_timestamp,
    ft.tx_index
  from solana.core.fact_transactions ft
  join date_filter df 
    on ft.block_timestamp between df.start_date and df.end_date
  where 
    ft.succeeded = true
),

price_data as(
  select 
    hour,
    token_address,
    symbol,
    price
  from solana.price.ez_prices_hourly ph
  join date_filter df 
    on ph.hour between df.start_date and df.end_date
  where 
    ph.blockchain = 'solana'
),

swap_list as(
  select 
    fs.block_timestamp,
    fs.tx_id,
    fs.block_id,
    fs.swapper,
    fs.program_id,
    fs.swap_program,
    fs.swap_to_mint,
    fs.swap_to_amount,
    fs.swap_from_mint,
    fs.swap_from_amount
  from solana.defi.fact_swaps fs
  join date_filter df 
    on fs.block_timestamp between df.start_date and df.end_date
  where 
    fs.succeeded = true
    and fs.swap_from_mint != fs.swap_to_mint
),

combined_swaps as (
  select 
    sl.*,
    t.tx_index,
    pd.price
  from swap_list sl
  join tx_list t 
    on sl.tx_id = t.tx_id
  join price_data pd
    on pd.hour = date_trunc('hour', sl.block_timestamp)
    and pd.token_address = sl.swap_from_mint
)

select 
  count(s1.tx_id) as "N° of Sandwiches",
  sum((s3.swap_to_amount - s1.swap_from_amount) * s1.price) as "Sandwich PnL [USD]"
from combined_swaps s1
join combined_swaps s2 
  on s1.block_id = s2.block_id 
  and s1.program_id = s2.program_id
  and s1.swap_to_mint = s2.swap_to_mint
  and s1.swap_from_mint = s2.swap_from_mint
  and s2.tx_index = s1.tx_index+1
  and s2.swapper <> s1.swapper
join combined_swaps s3 
  on s1.block_id = s3.block_id 
  and s1.tx_id <> s3.tx_id
  and s1.program_id = s3.program_id
  and s1.swap_to_mint = s3.swap_from_mint
  and s1.swap_from_mint = s3.swap_to_mint
  and s1.swap_to_amount = s3.swap_from_amount
  and s2.swapper <> s3.swapper
  and s3.tx_index = s2.tx_index+1
  and s3.swap_to_amount > s1.swap_from_amount

WITH token_holdings AS (
  SELECT block_timestamp, post_tokens.value:owner AS wallet, post_tokens.value:mint AS token, post_tokens.value:uiTokenAmount:uiAmount AS amount,
    ROW_NUMBER() OVER (PARTITION BY wallet, token ORDER BY block_timestamp DESC) AS rn
  FROM solana.core.fact_transactions, LATERAL FLATTEN(input => post_token_balances) post_tokens
  WHERE SUCCEEDED = 'True'
  AND post_tokens.value:owner = wallet
  AND post_tokens.value:mint = 'bSo13r4TkiE4KumL71LsHTPpL2euBYLFx6h9HP3piy1'
  AND block_timestamp >= CURRENT_DATE - 90
  ORDER BY wallet, token, rn
)
, base as (select wallet, amount
    FROM token_holdings 
     WHERE rn = 1 AND amount IS NOT NULL AND amount <> ''
      and amount > 0.1
     order by amount desc
  )

select
  	      case 
          -- when amount < 1 then 'Hold < 1 BSOL'
          when amount < 10 then 'Hold < 10 BSOL'
          when amount < 100 then 'Hold < 100 BSOL'
          when amount < 1000 then 'Hold < 1k BSOL'
          when amount < 10000 then 'Hold < 10k BSOL'
  			else 'Hold > 10k BSOL' end as type,
    -- https://count.co/sql-resources/snowflake/histograms
    count(wallet) as user_count,
    floor(amount / 10) * 10 as bin_start,
    floor(amount / 10) * 10 + 10 as bin_end,
    concat(
        cast(floor(amount / 10) * 10 as STRING),
        ' - ',
        cast(floor(amount / 10) * 10 + 10 as STRING)
    ) as bin_range,
    sum(user_count) over (order by bin_start) as cumu_address
from
    base
group by
    bin_start,
    bin_end,
    bin_range,
  	type
order by
    bin_start
limit 2500

with bonk_price as (
  select hour,
    symbol,
    token_address,
    price
  from solana.price.ez_prices_hourly
  where token_address = 'DezXAZ8z7PnrnRJjz3wXBoRgixCa6xjnB7YaB1pPB263'
)
, base as (
select a.block_timestamp,
    a.tx_id,
    swapper as user_address,
    swap_from_mint,
    swap_from_amount,
    case when c.symbol is null then 'Unknown' else c.symbol end as swap_from_symbol,
    swap_to_amount as bonk_amount,
    swap_to_amount * price as volume_bonk_usd,
    amount * 3 as fee_amount, -- Fees split into 3 addresses equally, so we just multiply by 3
    amount * 3 * price as fee_amount_usd,
    0.00203928 as rent_returned
from solana.defi.fact_swaps_jupiter_summary a
join bonk_price on (date_trunc('hour', a.block_timestamp) = hour)
join (select * from solana.core.fact_transfers 
    where tx_to = 'CbX9HYvyHBa2RuotGN8Y8hCKow6xppdrhAh6RL6i2BEf'  -- Fee collected address (one of the three)
      and mint = 'DezXAZ8z7PnrnRJjz3wXBoRgixCa6xjnB7YaB1pPB263'
      and block_timestamp >= '2024-01-12') b on (swapper = tx_from and a.tx_id = b.tx_id)
left join solana.price.ez_asset_metadata c on (swap_from_mint = c.token_address)
where swap_to_mint = 'DezXAZ8z7PnrnRJjz3wXBoRgixCa6xjnB7YaB1pPB263'
  and a.block_timestamp >= '2024-01-12'
-- order by block_timestamp desc limit 10
)

  swap_from_mint,
  count(distinct tx_id) as tx_count,
  count(distinct user_address) as user_count,
  round(sum(volume_bonk_usd),2) as bonk_scoop_usd,

      val.validator_name as validator
      --vpk.pubkey as validator

    from solana.gov.fact_rewards_fee rf
    inner join solana.gov.fact_validators val 
          on rf.pubkey = val.node_pubkey
          and rf.epoch_earned = val.epoch
    inner join validators_pubkeys vpk on vpk.pubkey = val.node_pubkey
    join date_filter
      on rf.block_timestamp between start_date and end_date
     
)

  from leader_slots ls
  left join solana.core.fact_transfers ft
    on ls.block_id = ft.block_id
  join date_filter
    on ft.block_timestamp between start_date and end_date

,jito_tips_counter as(
  select 
    dt,
    validator,
    block_id,
    sum(case when tx_id is not null then 1 else 0 end) as n_tips

  from jito_tips
  group by 1,2,3

)

,grouped_data as(
  select 
    dt,
    validator,
    percentile_cont(0.25) within group (order by n_tips) as p25,

Monthly Metaplex NFT Mints

WITH price AS (
SELECT
  hour::date AS date1,
  CASE
    WHEN token_address = 'So11111111111111111111111111111111111111112' THEN 'So11111111111111111111111111111111111111111'
    ELSE token_address
  END AS token_address,
  AVG(price) AS price
FROM solana.price.ez_prices_hourly
  -- WHERE token_address = ''
GROUP BY 1,2
),

base AS (
SELECT
  block_timestamp,
  tx_id,
  mint,
  purchaser,
  mint_price * price AS mint_price,
  mint_currency,
  program_id
FROM solana.nft.fact_nft_mints m JOIN price p 
ON m.block_timestamp::date = p.date1 AND m.mint_currency = p.token_address
  WHERE program_id IN (
  'BGUMAp9Gq7iTEuizy4pqaxsTyUCBK68MDfK752saRPUY',
  'CndyV3LdqHUfDLmE5naZjVN8rBZz4tqhdefbAnjHG3JR',
  'metaqbxxUerdq28cj1RbAWkYQm3ybzjb6a8bt518x1s'
)
  AND succeeded = 'TRUE'
-- having mint_price < 1 --excluding liquidity pool nft mint prices
)

SELECT
  DATE_TRUNC(month, block_timestamp) AS timespan,
  COUNT(DISTINCT tx_id) AS mint_transactions,
  SUM(mint_transactions) OVER (ORDER BY timespan) AS cummul_mint_transactions,
  COUNT(DISTINCT mint) AS NFT_mints,
  SUM(NFT_mints) OVER (ORDER BY timespan) AS cummul_NFT_mints,
  COUNT(DISTINCT purchaser) AS minters,
  SUM(minters) OVER (ORDER BY timespan) AS cummul_minters,
  ROUND(SUM(
      CASE
        WHEN mint_price < 1 THEN mint_price
        ELSE 0
      END), 2) AS total_mint_price,
  SUM(total_mint_price) OVER (ORDER BY timespan) AS cummul_mint_price,
  ROUND(AVG(
      CASE
        WHEN mint_price < 1 THEN mint_price
        ELSE 0
      END), 7) AS avg_mint_price
FROM base
GROUP BY 1--, 2 
ORDER BY timespan

WITH price AS (

    price
  FROM
    solana.price.ez_prices_hourly
  WHERE
    token_address = 'So11111111111111111111111111111111111111112'
  ORDER BY
    hour DESC
  LIMIT 1
),

nft_data AS (
  SELECT
    SUM(CASE WHEN seller = '{{wallet}}' THEN sales_amount ELSE 0 END) AS total_sells_amount,
    SUM(CASE WHEN purchaser = '{{wallet}}' THEN sales_amount ELSE 0 END) AS total_buys_amount,
    COUNT(CASE WHEN seller = '{{wallet}}' THEN tx_id END) AS total_sells_count,
    COUNT(CASE WHEN purchaser = '{{wallet}}' THEN tx_id END) AS total_buys_count
  FROM
    solana.nft.fact_nft_sales
  WHERE
    (seller = '{{wallet}}' OR purchaser = '{{wallet}}')
    AND (marketplace ILIKE '%magic eden%' OR marketplace ILIKE '%magic eden v2%')
    AND succeeded
)

SELECT
  nft_data.total_sells_amount,
  nft_data.total_sells_amount * p.price AS sell_volume,
  nft_data.total_buys_amount,
  nft_data.total_buys_amount * p.price AS buy_volume,
  nft_data.total_sells_count,
  nft_data.total_buys_count
FROM 
  nft_data
JOIN 
  price p
ON 
  TRUE;  -- Cross join with price CTE

WITH swap_data AS (
  SELECT
    swapper,
    SUM(SWAP_FROM_AMOUNT_USD) as total
  FROM
    solana.defi.ez_dex_swaps
  WHERE
    swap_program like 'jupiter%'
    AND swapper = '{{Address}}'
  GROUP BY
    swapper
)
SELECT
  '{{Address}}' as swapper,
  COALESCE(total, 0) as total
FROM
  swap_data
RIGHT JOIN
  (SELECT '{{Address}}' as swapper) as addr
ON
  swap_data.swapper = addr.swapper;
